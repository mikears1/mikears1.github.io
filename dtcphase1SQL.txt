create schema dtc;

use dtc;


-- *** Table Creation ***

-- Dentist table stores dentist basic information
create table Dentist(
  Dentist_ID integer not null auto_increment,
  Dentist_First_Name varchar(35) not null,
  Dentist_Middle_Name varchar(35),
  Dentist_Last_Name varchar(35) not null,
  Dentist_Join_Date DATE not null,
  constraint Dentist_PK primary key (Dentist_ID));

-- Account table stores balances for patient accounts
create table Account(
  Account_ID integer not null auto_increment,
  Account_Balance DECIMAL (10,2) not null,
  constraint Account_PK primary key (Account_ID));

-- Payment table: Stores payment records (date, amount, patient account)
create table Payment(
  Payment_ID integer not null auto_increment,
  Payment_Date DATE not null,
  Payment_Amount DECIMAL (10,2) not null,
  Account_ID integer not null,
  constraint Payment_PK primary key (Payment_ID),
  constraint Payment_FK foreign key (Account_ID) references Account(Account_ID) on delete cascade);

-- Patient table: Stores patient personal information
create table Patient(
  Patient_ID integer not null auto_increment,
  Account_ID integer not null,
  Patient_First_Name varchar(35) not null,
  Patient_Middle_Name varchar(35) not null,
  Patient_Last_Name varchar(35) not null,
  Patient_Street varchar(35) not null,
  Patient_City varchar(35) not null,
  Patient_State varchar(2) not null,
  Patient_Zip varchar(5) not null,
  Patient_Phone_Primary varchar(15) not null,
  Patient_Phone_Secondary varchar(15) not null,
  Patient_SSN varchar(9) not null,
  Patient_DOB date not null,
  Patient_Minor boolean not null,
  Patient_Head_Of_House integer,
  Patient_Allergies varchar(35) not null,
  constraint Patient_PK primary key (Patient_ID),
  constraint Patient_FK foreign key (Account_ID) references Account(Account_ID) on delete cascade,
  constraint Patient_FK2 foreign key (Patient_Head_Of_House) references Patient(Patient_ID) on delete cascade);



-- Appointment table: Stores information regarding future appointments (dentist, patient, date/time) 
create table Appointment(
  Appointment_ID integer not null auto_increment,
  Dentist_ID integer not null,
  Patient_ID integer not null,
  Appointment_Date DATE not null,
  Appointment_Time TIME not null,
  constraint Appointment_PK primary key (Appointment_ID),
  constraint Appointment_FK1 foreign key (Dentist_ID) references Dentist (Dentist_ID) on delete cascade,
  constraint Appointment_FK2 foreign key (Patient_ID) references Patient (Patient_ID) on delete cascade);


-- Service table: Stores information regarding services provided (code, description, cost) 
create table Service(
  Service_ID integer not null auto_increment,
  Service_Description varchar(100) not null,
  Service_Cost DECIMAL (10,2) not null,
  constraint Service_PK primary key (Service_ID));

-- Visit table: Stores information regarding previous visits (appointments attended) (dentist, patient, date/time) 
create table Visit(
  Visit_ID integer not null auto_increment,
  Dentist_ID integer not null,
  Patient_ID integer not null,
  Visit_Date DATE not null,
  Visit_Time TIME not null,
  constraint Visit_PK primary key (Visit_ID),
  constraint Visit_FK1 foreign key (Dentist_ID) references Dentist(Dentist_ID) on delete cascade,
  constraint Visit_FK2 foreign key (Patient_ID) references Patient(Patient_ID) on delete cascade);

-- Scheduled service table: Links future appointments to the service that is scheduled to be provided in the appointment.
create table Scheduled_Service(
  Scheduled_Service_ID integer not null auto_increment,
  Appointment_ID integer not null,
  Service_ID integer not null,
  constraint Scheduled_Service_PK primary key (Scheduled_Service_ID),
  constraint Scheduled_Service_FK1 foreign key (Appointment_ID) references Appointment(Appointment_ID) on delete cascade,
  constraint Scheduled_Service_FK2 foreign key (Service_ID) references Service(Service_ID) on delete cascade); 
  
-- Completed service table: Links past visits to the service that was provided in the visit  
create table Completed_Service(
  Completed_Service_ID integer not null auto_increment,
  Service_ID integer not null,
  Visit_ID integer not null,
  constraint Completed_Service_PK primary key (Completed_Service_ID),
  constraint Completed_Service_FK1 foreign key (Service_ID) references Service(Service_ID) on delete cascade,
  constraint Completed_Service_FK2 foreign key (Visit_ID) references Visit(Visit_ID) on delete cascade);
